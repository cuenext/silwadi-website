          def replace_featured(path, replacement):
              p = Path(path)
              text = p.read_text(encoding='utf-8')
              pattern = r'<div class="featured-doctors">.*?</div><div class="section-link reveal">'
              new_text, n = re.subn(pattern, replacement + '<div class="section-link reveal">', text, count=1, flags=re.S)
              assert n == 1, f'Expected one featured-doctors section in {path}, got {n}'
              p.write_text(new_text, encoding='utf-8')

          replace_featured('index.html', en_cards)

          css_path = Path('styles.css')
