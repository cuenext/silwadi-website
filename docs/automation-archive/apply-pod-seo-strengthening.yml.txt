          from pathlib import Path

          pod_path = Path('treatments/people-of-determination.html')
          pod = pod_path.read_text(encoding='utf-8')

              raise SystemExit('Could not find unique POD SEO schema insertion marker')

          css_marker = '    .pod-consultation{padding:34px 0 86px}.pod-consultation__box{background:var(--pod-deep);color:#fff;padding:50px 54px;display:flex;align-items:center;justify-content:space-between;gap:38px}.pod-consultation__box h2{margin:0;font-size:35px;line-height:1.08;letter-spacing:-.035em}.pod-consultation__box p{margin:10px 0 0;color:#c5d9dd;font-size:13px;line-height:1.65;max-width:650px}.pod-consultation__box .pod-btn{background:#fff;color:var(--pod-navy);flex:0 0 auto}\n'
          faq_css = css_marker + '''    .pod-faq{background:#fff}.pod-faq__intro{max-width:760px;margin-bottom:34px}.pod-faq__intro .pod-copy{margin:16px 0 0}.pod-faq__grid{display:grid;grid-template-columns:repeat(2,minmax(0,1fr));border-top:1px solid var(--pod-line)}.pod-faq__item{padding:28px 28px 28px 0;border-bottom:1px solid var(--pod-line)}.pod-faq__item:nth-child(even){padding-left:28px;border-left:1px solid var(--pod-line)}.pod-faq__item h3{margin:0 0 10px;color:var(--pod-navy);font-size:17px;line-height:1.35}.pod-faq__item p{margin:0;color:var(--pod-muted);font-size:13px;line-height:1.7}\n'''

              raise SystemExit('Could not find POD subnav marker')

          if pod.count(consultation_marker) != 1:
              raise SystemExit('Could not find POD consultation section marker')
          pod = pod.replace(consultation_marker, faq_section, 1)
          pod_path.write_text(pod, encoding='utf-8')

          locations_path = Path('locations.html')
          locations = locations_path.read_text(encoding='utf-8')
          old_actions = '<div class="location-actions"><a class="btn btn--primary" href="contact.html#consultation-form">Book a Consultation</a><a class="btn btn--secondary" href="tel:+97126662408">Call Al Raha</a><a class="btn btn--secondary" href="https://www.google.com/maps/search/?api=1&amp;query=Dr%20Munir%20Silwadi%20Dental%20Centre%20Al%20Raha%20Mall%20Abu%20Dhabi">Get Directions</a></div>'
          new_actions = '<div class="location-actions"><a class="btn btn--primary" href="contact.html#consultation-form">Book a Consultation</a><a class="btn btn--secondary" href="treatments/people-of-determination.html">People of Determination Care</a><a class="btn btn--secondary" href="tel:+97126662408">Call Al Raha</a><a class="btn btn--secondary" href="https://www.google.com/maps/search/?api=1&amp;query=Dr%20Munir%20Silwadi%20Dental%20Centre%20Al%20Raha%20Mall%20Abu%20Dhabi">Get Directions</a></div>'
          if locations.count(old_actions) != 1:
              raise SystemExit('Could not find unique Al Raha action block')
          locations = locations.replace(old_actions, new_actions, 1)
          locations_path.write_text(locations, encoding='utf-8')

          sitemap_path = Path('sitemap.xml')
          sitemap = sitemap_path.read_text(encoding='utf-8')
          old_sitemap = '    <loc>https://silwadi.ae/treatments/people-of-determination.html</loc>\n    <lastmod>2026-09-16</lastmod>'
          new_sitemap = '    <loc>https://silwadi.ae/treatments/people-of-determination.html</loc>\n    <lastmod>2026-09-17</lastmod>'
          if sitemap.count(old_sitemap) != 1:
              raise SystemExit('Could not find POD sitemap entry dated 2026-09-16')
          sitemap = sitemap.replace(old_sitemap, new_sitemap, 1)
          sitemap_path.write_text(sitemap, encoding='utf-8')
          PY
      - name: Run POD SEO regression contract
