name: Remove about gallery

on:
  push:
    branches:
      - main
    paths:
      - .github/workflows/remove-about-gallery.yml
  workflow_dispatch:

permissions:
  contents: write

jobs:
  remove:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Remove Inside Silwadi gallery block
        run: |
          python3 <<'PY'
          from pathlib import Path
          import re

          path = Path('about.html')
          html = path.read_text(encoding='utf-8')

          pattern = r'\n\s*<section class="institutional-section about-gallery-section">[\s\S]*?\n\s*</section>\n'
          updated, count = re.subn(pattern, '\n', html, count=1)

          if count != 1:
              raise SystemExit(f'Expected to remove 1 about-gallery-section, removed {count}')

          path.write_text(updated, encoding='utf-8')
          PY

          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add about.html
          git commit -m "Remove About page inside gallery section" || exit 0
          git push
